# Q2: Write a program in python to implement Fibonacci series up to user entered number (Use recursive Function)

def fibonacci(n):
    if n <= 1:
        return n
    else:
        return fibonacci(n-1) + fibonacci(n-2)

num = int(input("Enter number of terms: "))
print("Fibonacci Series:")
for i in range(num):
    print(fibonacci(i), end=" ")

    print("Result:", add(a, b))
elif op == '-':
    print("Result:", sub(a, b))
elif op == '*':
    print("Result:", mul(a, b))
elif op == '/':
    print("Result:", div(a, b))

    print("Invalid operator!")

# Q5: Write a program to demonstrate local variable, global variable and global keyword

x = 10  # global variable

def show():
    y = 5  # local variable
    print("Inside function, local y =", y)
    print("Inside function, global x =", x)

def change():
    global x
    x = x + 5
    print("Global x changed to:", x)

show()
change()
print("Outside function, x =", x)

# Q6: Write a Python function that accepts a string and counts the number of upper and lower case letters

def count_case(s):
    upper = sum(1 for c in s if c.isupper())
    lower = sum(1 for c in s if c.islower())
    print("Uppercase letters:", upper)
    print("Lowercase letters:", lower)

text = input("Enter a string: ")
count_case(text)

# Q8: Write a Python function that takes a number as a parameter and checks whether the number is prime or not

def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n/2)+1):
        if n % i == 0:
            return False
    return True

num = int(input("Enter a number: "))
if is_prime(num):
    print(num, "is a Prime Number")
else:
    print(num, "is not a Prime Number")

# Q9: Write a Python function to print the even and odd numbers from a given list

def even_odd(lst):
    even = [x for x in lst if x % 2 == 0]
    odd = [x for x in lst if x % 2 != 0]
    print("Even numbers:", even)
    print("Odd numbers:", odd)

numbers = list(map(int, input("Enter numbers separated by space: ").split()))
even_odd(numbers)

# Q10: Write a Python program to create Fibonacci series up to n using Lambda

from functools import reduce

n = int(input("Enter number of terms: "))
fibo = lambda n: reduce(lambda x, _: x + [x[-1] + x[-2]], range(n-2), [0, 1])
print("Fibonacci Series:", fibo(n))

# Q12: Write a Python program to calculate the sum of the positive and negative numbers of a given list of numbers using the lambda function

from functools import reduce

numbers = list(map(int, input("Enter numbers separated by space: ").split()))

pos_sum = reduce(lambda x, y: x + y, filter(lambda n: n > 0, numbers))
neg_sum = reduce(lambda x, y: x + y, filter(lambda n: n < 0, numbers))

print("Sum of positive numbers:", pos_sum)
print("Sum of negative numbers:", neg_sum)

# Q13: Write a Python program to remove None values from a given list using the lambda function

# Q15: Write a python program to calculate gross salary and net salary of an employee. create separate functions to calculate DA, HRA, PF and Tax

def da(basic): return basic * 0.10
def hra(basic): return basic * 0.15
def pf(basic): return basic * 0.05
def tax(basic): return basic * 0.08

basic = float(input("Enter Basic Salary: "))

gross = basic + da(basic) + hra(basic)
net = gross - (pf(basic) + tax(basic))

print("Gross Salary:", gross)
print("Net Salary:", net)
